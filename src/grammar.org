* grammar for the scas
program ::= statement* EOF
program = Vec<Statement>

statement ::= declaration | assignment_stmt | print_stmt | return_stmt |
if_stmt | while_stmt | for_stmt | break_stmt | continue_stmt | expr_stmt | block

block ::= “{” statement* “}”

declaration ::= variable_decl | function_decl

variable_decl ::= “let” IDENTIFIER (“=” expression)? “;”

function_decl ::= IDENTIFIER “(” parameter_list? “)” “=” expression

parameter_list ::= IDENTIFIER (“,” IDENTIFIER)*

assignment_stmt ::= IDENTIFIER assignment_operator expression “;”

assignment_operator ::= “=” | “+=” | “-=” | “*=” | “/=” | “%=” | “^=” |
“&=” | “|=” | "~="

print_stmt ::= “print” “(” expression “)” “;”

return_stmt ::= “return” expression? “;”

break_stmt ::= “break” “;”

continue_stmt ::= “continue” “;”

if_stmt ::= “if” expression block (“else” block)?

while_stmt ::= “while” expression block

for_stmt ::= “for” for_initializer “,” expression “,” assignment_expr
block

for_initializer ::= “let” IDENTIFIER “=” expression | assignment_expr

expression ::= logical_or

logical_or ::= logical_and | logical_or “or” logical_and

logical_and ::= equality | logical_and “and” equality

equality ::= comparison | equality “==” comparison | equality “!=”
comparison

comparison ::= bitwise_or | comparison “<” bitwise_or | comparison “<=”
bitwise_or | comparison “>” bitwise_or | comparison “>=” bitwise_or

bitwise_or ::= bitwise_xor | bitwise_or “|” bitwise_xor

bitwise_xor ::= bitwise_and | bitwise_xor “xor” bitwise_and

bitwise_and ::= shift | bitwise_and “&” shift

shift ::= additive | shift “<<” additive | shift “>>” additive

additive ::= multiplicative | additive “+” multiplicative | additive “-”
multiplicative

multiplicative ::= exponent | multiplicative “*” exponent |
multiplicative “/” exponent | multiplicative “%” exponent

exponent ::= unary | unary “^” exponent

unary ::= “!” unary | “-” unary | “~” unary | postfix

postfix ::= primary (“(” argument_list? “)”)*

argument_list ::= expression (“,” expression)*

primary ::= NUMBER | STRING | IDENTIFIER | “true” | “false” | “nil” |
“(” expression “)”

assignment_expr ::= IDENTIFIER assignment_operator expression

expr_stmt ::= expression “;”
