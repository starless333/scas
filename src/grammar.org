(* ========================================== *)
(*              PROGRAM & STATEMENTS          *)
(* ========================================== *)

program         ::= statement* EOF

statement       ::= declaration 
                  | assignment_stmt 
                  | print_stmt 
                  | return_stmt 
                  | if_stmt 
                  | while_stmt 
                  | for_stmt 
                  | break_stmt 
                  | continue_stmt 
                  | expr_stmt 
                  | doc_comment
                  | render_block
                  | render_line
                  | show_stmt
                  | block

render_block    ::= "\\{" statement* "\\}"

render_line     ::= "\\[" expression ("=>")? "\\]"

show_stmt       ::= expression "=>" ";"

doc_comment     ::= "///" DOC_TEXT

block           ::= "{" statement* "}"

(* ========================================== *)
(*                DECLARATIONS                *)
(* ========================================== *)

declaration     ::= variable_decl | function_decl | math_function_decl

variable_decl   ::= "let" IDENTIFIER ("=" expression)? ";"

function_decl   ::= "fn" IDENTIFIER "(" parameter_list? ")"
                    ("=" expression ";" | block)

math_function_decl ::= IDENTIFIER "(" parameter_list? ")" "=" expression ";"

parameter_list  ::= IDENTIFIER ("," IDENTIFIER)*

(* ========================================== *)
(*               CONTROL FLOW                 *)
(* ========================================== *)

if_stmt         ::= "if" expression block ("else" (if_stmt | block))?

while_stmt      ::= "while" expression block

(* The two separators are semicolons. The update has no trailing semicolon. *)
for_stmt        ::= "for" for_initializer ";" expression ";" for_update block

for_initializer ::= "let" IDENTIFIER ("=" expression)? | assignment_expr | expression

for_update     ::= assignment_expr | expression

(* ========================================== *)
(*              OTHER STATEMENTS              *)
(* ========================================== *)

assignment_stmt ::= assignment_expr ";"

assignment_expr ::= IDENTIFIER assignment_operator expression

assignment_operator ::= "=" | "+=" | "-=" | "*=" | "/=" | "%=" | "^=" | "&=" | "|="

print_stmt      ::= "print" "(" expression ")" ";"

return_stmt     ::= "return" expression? ";"

break_stmt      ::= "break" ";"

continue_stmt   ::= "continue" ";"

expr_stmt       ::= expression ";"

(* ========================================== *)
(*                EXPRESSIONS                 *)
(* (Ordered lowest to highest precedence)    *)
(* ========================================== *)

expression      ::= logical_or

logical_or      ::= logical_and ("or" logical_and)*

logical_and     ::= equality ("and" equality)*

equality        ::= comparison (("==" | "!=") comparison)*

comparison      ::= bitwise_or (("<" | "<=" | ">" | ">=") bitwise_or)*

bitwise_or      ::= bitwise_xor ("|" bitwise_xor)*

bitwise_xor     ::= bitwise_and ("xor" bitwise_and)*

bitwise_and     ::= shift ("&" shift)*

shift           ::= additive (("<<" | ">>") additive)*

additive        ::= multiplicative (("+" | "-") multiplicative)*

multiplicative  ::= unary (("*" | "/" | "%") unary | implicit_product)*

implicit_product ::= exponent

unary           ::= ("!" | "-" | "~") unary | exponent

exponent        ::= postfix ("^" exponent)?

postfix         ::= primary ("(" argument_list? ")" | "[" index_list "]")*

index_list      ::= expression ("," expression)*

argument_list   ::= expression ("," expression)*

array_elements  ::= expression ("," expression)* ","?

matrix_literal  ::= "[" matrix_row (";" matrix_row)+ "]"

matrix_row      ::= expression ("," expression)*

primary         ::= NUMBER 
                  | STRING 
                  | IDENTIFIER 
                  | "[" array_elements? "]"
                  | matrix_literal
                  | "true" 
                  | "false" 
                  | "nil" 
                  | "(" expression ")"
